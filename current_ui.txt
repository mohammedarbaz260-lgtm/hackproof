<script>
async function sendMessage() {

    const input = document.getElementById("question");
    const question = input.value.trim();

    if (!question) return;

    const chat = document.getElementById("chat");

    chat.innerHTML += `
        <div class="message user">
            <b>You:</b> ${question}
        </div>
    `;

    input.value = "";

    try {

        const response = await fetch("/chat", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                question: question
            })
        });

        const data = await response.json();

        chat.innerHTML += `
            <div class="message bot">
                <b>HackProof:</b><br>
                ${data.answer}
            </div>
        `;

    } catch (error) {

        chat.innerHTML += `
            <div class="message bot">
                <b>HackProof:</b><br>
                Error connecting to HackProof server.
            </div>
        `;
    }

    chat.scrollTop = chat.scrollHeight;
}

function clearChat() {
    const chat = document.getElementById("chat");
    chat.innerHTML = `
        <div class="message bot">
            <b>HackProof:</b><br>
            Hello! I am HackProof AI. Ask me a cybersecurity question.
        </div>
    `;
}

MEMORY_FILE = Path("memory.txt")

def load_memory():
    if not MEMORY_FILE.exists():
        return ""
    return MEMORY_FILE.read_text(encoding="utf-8")[-12000:]

def save_memory(question, answer):
    entry = f"\nUser: {question}\nHackProof: {answer}\n"
    with MEMORY_FILE.open("a", encoding="utf-8") as f:
        f.write(entry)
